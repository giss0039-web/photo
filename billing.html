const text = ret.data.text; // 1. 自動抓取日期 const dates = text.match(/\d{4}[-/年]\d{1,2}[-/月]\d{1,2}/g); if (dates && dates.length > 0) { const formattedDate = dates[0].replace(/[年/]/g, '-').replace('月', '-').replace('日', ''); document.getElementById('g_date').value = formattedDate; } // 2. 智慧抓取金額（優先搜尋「總計」關鍵字，若無則排除大於 10000 的統編或隨機碼） let targetAmount = ''; const totalMatch = text.match(/總\s*計\D*(\d+)/); if (totalMatch && totalMatch[1]) { targetAmount = totalMatch[1]; } else { const amounts = text.match(/\d+/g); if (amounts) { // 過濾掉像統編8碼、隨機碼4碼等不合理的數字，抓取 1 ~ 5000 之間的常見消費金額 const validNums = amounts.map(Number.parseFloat).filter(n => n > 0 && n < 5000); if (validNums.length > 0) { targetAmount = validNums[validNums.length - 1]; // 通常總計在發票中後段 } } } if (targetAmount) { document.getElementById('g_price').value = targetAmount; document.getElementById('g_qty').value = 1; calcGeneralTotal(); } // 3. 自動抓取店家名稱（如：歐特儀） if (text.includes('歐特儀')) { document.getElementById('g_vendor').value = '歐特儀（擎天岡停車場）'; } else { const lines = text.split('\n').filter(l => l.trim() !== ''); if (lines.length > 0) { document.getElementById('g_vendor').value = lines[0].substring(0, 20); } }
